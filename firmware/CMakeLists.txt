cmake_minimum_required(VERSION 3.20)
list(APPEND BOARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(zigmodem_firmware)
add_subdirectory(../vendor ${CMAKE_CURRENT_BINARY_DIR}/vendor)
target_sources(app PRIVATE src/main.c src/identity.c ../core/modem.c ../core/secure.c ../core/codec.c)
target_link_libraries(app PRIVATE zm_noise zm_heatshrink)
target_include_directories(app PRIVATE ../core)
