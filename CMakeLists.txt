cmake_minimum_required(VERSION 3.20)
project(zigmodem VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_subdirectory(vendor)
option(ZM_CURVE32 "Exercise the microcontroller Curve25519 implementation on the host" OFF)
if(ZM_CURVE32)
  target_compile_definitions(zm_noise PRIVATE ZIGMODEM_TEST_32BIT_CURVE=1)
endif()
add_library(zigmodem core/modem.c)
target_include_directories(zigmodem PUBLIC core)
add_library(zigmodem_secure core/secure.c core/codec.c)
target_include_directories(zigmodem_secure PUBLIC core)
target_link_libraries(zigmodem_secure PUBLIC zm_noise zm_heatshrink)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(zigmodem PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(zigmodem_secure PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(test_modem tests/test_modem.c)
target_link_libraries(test_modem PRIVATE zigmodem)
add_test(NAME modem COMMAND test_modem)
add_executable(test_secure tests/test_secure.c)
target_link_libraries(test_secure PRIVATE zigmodem_secure zigmodem)
add_test(NAME secure COMMAND test_secure)
add_executable(test_crypto tests/test_crypto.c)
target_link_libraries(test_crypto PRIVATE zigmodem_secure)
add_test(NAME crypto COMMAND test_crypto)
if(UNIX)
  add_executable(zigmodem-sim host/sim.c)
  target_link_libraries(zigmodem-sim PRIVATE zigmodem)
endif()
